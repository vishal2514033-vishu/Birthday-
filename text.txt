<script>

function startSurprise() {

  document.getElementById("birthday").classList.remove("hidden");
  document.getElementById("photos").classList.remove("hidden");
  document.getElementById("loveletter").classList.remove("hidden");
  document.getElementById("final").classList.remove("hidden");

  document.querySelector(".hero button").textContent =
    "Your Surprise Is Opened ❤️";

  document.querySelector(".hero button").disabled = true;

  document.getElementById("birthday").scrollIntoView({
    behavior: "smooth"
  });

  hearts();
}

function hearts() {

  const symbols = ["❤️", "💗", "💕", "💖", "🌸"];

  function addHeart() {

    const heart = document.createElement("div");

    heart.className = "heart";
    heart.textContent =
      symbols[Math.floor(Math.random() * symbols.length)];

    heart.style.left = Math.random() * 100 + "vw";
    heart.style.fontSize = (18 + Math.random() * 22) + "px";

    document.body.appendChild(heart);

    setTimeout(() => heart.remove(), 6000);
  }

  for (let i = 0; i < 15; i++) {
    setTimeout(addHeart, i * 150);
  }

  setInterval(addHeart, 500);
}
